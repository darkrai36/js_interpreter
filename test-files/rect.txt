function draw(width, height) {
    if (width < 2) {
        return "width must be >= 2";
    }
    if (height < 2) {
        return "height must be >= 2";
    }

    let result = "";

    result = result + "*";
    let i = 0;
    while (i < width - 2) {
        result = result + "-";
        i = i + 1;
    }
    result = result + "*\n";

    let row = 0;
    while (row < height - 2) {
        result = result + "|";
        let j = 0;
        while (j < width - 2) {
            result = result + " ";
            j = j + 1;
        }
        result = result + "|\n";
        row = row + 1;
    }

    if (height > 1) {
        result = result + "*";
        let k = 0;
        while (k < width - 2) {
            result = result + "-";
            k = k + 1;
        }
        result = result + "*";
    }

    return result;
}

print(draw(5, 5));
print("---");
print(draw(8, 4));
