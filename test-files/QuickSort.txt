function quickSort(arr) {
    let len = arr.length;

    if (len <= 1) {
        return arr;
    }

    let pivot = arr[0];
    let left = [];
    let right = [];

    for (let i = 1; i < len; i = i + 1) {
        if (arr[i] < pivot) {
            left.push(arr[i]);
        } else {
            right.push(arr[i]);
        }
    }

    let sortedLeft = quickSort(left);
    let sortedRight = quickSort(right);

    let result = [];

    for (let i = 0; i < sortedLeft.length; i = i + 1) {
        result.push(sortedLeft[i]);
    }

    result.push(pivot);

    for (let i = 0; i < sortedRight.length; i = i + 1) {
        result.push(sortedRight[i]);
    }

    return result;
}

let data = [64, 34, 25, 12, 22, 11, 90, 5];
print("Starting array:");
print(data);

let sortedData = quickSort(data);

print("Array after sorting (QuickSort):");
print(sortedData);
